## Installation

```bash
bunx dreamy add select
```

## Usage

Pass options as an `items` array on `Select.Root`. Each item requires `value` and `label`.

<Wrapper>
	<Select.Root
        width="xs"
        items={[
            { value: "strawberry", label: "Strawberry" },
            { value: "banana", label: "Banana" },
            { value: "orange", label: "Orange" },
        ]}
    >
        <Select.Trigger placeholder="Select a favorite fruit" />
        <Select.Content />
    </Select.Root>
</Wrapper>

```tsx
const fruits = [
    { value: "strawberry", label: "Strawberry" },
    { value: "banana", label: "Banana" },
    { value: "orange", label: "Orange" },
];

<Select.Root items={fruits}>
    <Select.Trigger placeholder="Select a favorite fruit" />
    <Select.Content />
</Select.Root>
```

### Top-layer positioning and UI SDK props

Select dropdown behavior is powered by [UI SDK Select](https://ui-sdk.dev/docs/components/select) with Dreamy styling and top-layer positioning. Configure placement with `positioning` on `Select.Root` or `side`/`align`/`sideOffset` on `Select.Content`. Use `container` only when targeting a custom container such as a shadow root or iframe.

<Wrapper>
    <Select.Root
        width="xs"
        items={[
            { value: "strawberry", label: "Strawberry" },
            { value: "banana", label: "Banana" },
            { value: "orange", label: "Orange" },
        ]}
    >
        <Select.Trigger placeholder="Select a favorite fruit" />
        <Select.Content side="bottom" align="start" />
    </Select.Root>
</Wrapper>

```tsx
<Select.Root items={fruits}>
    <Select.Trigger placeholder="Select a favorite fruit" />

</Select.Root>
```

### Custom item rendering

Use the `renderItem` prop on `Select.Root` to customise how each option is displayed. The render function receives the full item type from your `items` array, so extra fields such as `icon` are type-safe.

<Wrapper>
    <ItemsSelect />
</Wrapper>

```tsx
const fruits = [
    { value: "cherry", label: "Cherry", icon: <LuCherry /> },
    { value: "banana", label: "Banana", icon: <LuBanana /> },
    { value: "orange", label: "Orange", icon: <LuCitrus /> },
];

<Select.Root
    items={fruits}
    renderItem={(item) => (
        <HStack gap={2}>
            {item.icon}
            <Text>{item.label}</Text>
        </HStack>
    )}
>
    <Select.Trigger placeholder="Select a favorite fruit" />
    <Select.Content />
</Select.Root>
```

### Controlled

Use `value` and `onValueChange` to control the selected option.

<Wrapper>
    <ControlledSelect />
</Wrapper>

```tsx
export function ControlledSelect() {
    const [value, setValue] = useState<string | null>("strawberry");

    const fruits = [
        { value: "strawberry", label: "Strawberry" },
        { value: "banana", label: "Banana" },
        { value: "orange", label: "Orange" },
    ];

    return (
        <Select.Root
            items={fruits}
            value={value}
            onValueChange={(next) => setValue(next)}
            width="xs"
        >
            <Select.Trigger placeholder="Select a favorite fruit" />
            <Select.Content />
        </Select.Root>
    );
}
```

### Async loading

You can fetch data when the Select is opened via `onOpenChange` and update the `items` array. Use `showItems={false}` on `Select.Content` to render custom UI (such as a loading spinner) instead of the default items list.

<Wrapper>
    <AsyncSelect />
</Wrapper>

```tsx
export function AsyncSelect() {
    const [isLoading, setIsLoading] = useState(false);
    const [items, setItems] = useState<{ value: string; label: string }[]>([]);

    function fetchFruits(open: boolean) {
        if (!open || items.length > 0) return;

        setIsLoading(true);
        fetch("/api/fake-select-data")
            .then((res) => res.json())
            .then((data: string[]) =>
                setItems(data.map((fruit) => ({ value: fruit, label: fruit })))
            )
            .finally(() => setIsLoading(false));
    }

    return (
        <Select.Root items={items} onOpenChange={fetchFruits} width="xs">
            <Select.Trigger placeholder="Select a favorite fruit" />
            <Select.Content showItems={!isLoading}>
                <Spinner color="primary" py={4} />
            </Select.Content>
        </Select.Root>
    );
}
```

### Virtualized

For better performance with large lists (100+ items), use `Select.VirtualContent` instead of `Select.Content`. It only renders visible items.

        <Select.VirtualContent />
    </Select.Root>
</Wrapper>

```tsx

}));

<Select.Root items={manyItems}>
    <Select.Trigger placeholder="Select a number" />
    <Select.VirtualContent />

```

#### Props

- `estimatedItemHeight` - Estimated height of each item in pixels. For different select sizes: `xs`: `26`, `sm`: `28`, `md`: `32`, `lg`: `40`. Default: `32`
- `overscan` - Number of items to render outside the visible area. Default: `5`
- `maxHeight` - Maximum height of the virtualized list container in pixels. Default: `300`

### Inside modals and popovers

No extra prop is needed inside Dreamy UI overlays. Keep the Select portal enabled so its `dropdown` scope (1000) nests under the parent scope and escapes clipped or scrolling content. In a Modal, it resolves to `calc(1400 + 1000)` automatically.

Later sibling Modals still cover earlier Modal subtrees, and same-layer dropdowns follow open order. Do not disable the portal or add a manual layer or z-index. For a native or third-party overlay, pass its non-scrolling root element via the `container` prop on `Select.Root` or `Select.Content`.

<Wrapper>
    <SelectInModal />
</Wrapper>

```tsx
const fruits = [
    { value: "cherry", label: "Cherry" },
    { value: "banana", label: "Banana" },
    { value: "orange", label: "Orange" },
];

export function SelectInModal() {
    const [open, setOpen] = useState(false);

    return (
        <>
            <Button onClick={() => setOpen(true)} variant="primary">
                Open modal
            </Button>
            <Modal.Root
                open={open}
                onOpenChange={(nextOpen) => setOpen(nextOpen)}
            >
                <Modal.Overlay />
                <Modal.Content>
                    <Modal.Header>Select in a modal</Modal.Header>
                    <Modal.CloseButton />
                    <Modal.Body>
                        <Select.Root
                            defaultValue="cherry"
                            items={fruits}
                            width="xs"
                        >
                            <Select.Trigger placeholder="Select a favorite fruit" />
                            <Select.Content />
                        </Select.Root>
                    </Modal.Body>
                    <Modal.Footer>
                        <Button onClick={() => setOpen(false)}>Close</Button>
                    </Modal.Footer>
                </Modal.Content>
            </Modal.Root>
        </>
    );
}
```

### Keep open on select

Single-select closes automatically when an option is selected, while `multiple` keeps the menu open. To keep a single-select open after selection, intercept `onOpenChange` and call `details.cancel()` when `details.reason === "select"` (or control `open` and ignore `"select"` reasons).

<Wrapper>
    <Select.Root
        width="xs"
        onOpenChange={(nextOpen, details) => {
            if (!nextOpen && details.reason === "select") {
                details.cancel();
            }
        }}
        items={[
            { value: "strawberry", label: "Strawberry" },
            { value: "banana", label: "Banana" },
            { value: "orange", label: "Orange" },
        ]}
    >
        <Select.Trigger placeholder="Select a favorite fruit" />
        <Select.Content />
    </Select.Root>
</Wrapper>

```tsx
<Select.Root
```
