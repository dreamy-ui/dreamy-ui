## Installation

```bash	
pnpm dreamy add accordion
```

## Usage

Each item needs a unique `value`. By default, only one panel can be open at a time.

<Wrapper>
  	<Accordion.Root>
		{["item-1", "item-2", "item-3"].map((value, index) => (
			<Accordion.Item key={value} value={value}>
				<Accordion.Trigger>Item {index + 1}</Accordion.Trigger>
				<Accordion.Content>Hi!</Accordion.Content>
			</Accordion.Item>
		))}
	</Accordion.Root>
</Wrapper>


### Controlled

Control expanded panels with `value` and `onValueChange`.

<Wrapper>
	<ControlledAccordion />
</Wrapper>

```tsx
export function ControlledAccordion() {
    const [value, setValue] = useState<string[]>(["item-1"]);

    return (
        <Accordion.Root
            value={value}
            onValueChange={(details) => setValue(details.value)}
        >
            {["item-1", "item-2", "item-3"].map((itemValue, index) => (
                <Accordion.Item key={itemValue} value={itemValue}>
                    <Accordion.Trigger>Item {index + 1}</Accordion.Trigger>
                    <Accordion.Content>Hi!</Accordion.Content>
                </Accordion.Item>
            ))}
        </Accordion.Root>
    );
}
```
