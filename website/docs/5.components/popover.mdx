## Installation

```bash
pnpm dreamy add popover
```

## Usage

Popover behavior (focus, dismiss, positioning, top-layer rendering) is provided by UI SDK Popover (`@ui-sdk/react/popover`). Dreamy owns styling, compound parts, and CSS enter/exit transitions on the content surface. Use `open`, `defaultOpen`, and `onOpenChange` for controlled state — `onOpenChange` receives `(open, details)` where `details.reason` explains the change and `details.cancel()` can veto it.

Floating content renders through the UI SDK popup, which positions itself against the trigger (or `Popover.Anchor`) and shows in the browser top layer. Set `hasArrow` on the root; the pointer is `Popover.Arrow`, a single element positioned by the popup and styled via the popover recipe.

When Popover opens, focus moves into the panel. When it closes, focus returns to the trigger. Set `hasArrow` on `Popover.Root` to render the pointer (optional `arrowSize` / `arrowShadowColor`).

Use `reduceMotion` on the root or rely on `DreamyProvider` to skip transitions. Content mounts when the popover opens; the deprecated `lazyMount={false}` maps to `keepMounted` on `Popover.Content` if you need it to stay in the DOM.

### Automatic overlay stacking

Popover content renders in the browser top layer via the UI SDK popup (`popover="manual"`), so it naturally stacks above page content and other overlays without a Dreamy [`Portal`](/docs/components/portal) or z-index workaround.

To render the panel inside a specific element instead, pass `container` to `Popover.Content` (or the deprecated `portalProps.containerRef` on `Popover.Root`, which only honors `containerRef`).

```tsx
const overlayRef = useRef<HTMLDivElement>(null);

// Attach overlayRef to the third-party overlay's non-scrolling root.
<Popover.Root>
    <Popover.Trigger>
        <Button>Open Popover</Button>
    </Popover.Trigger>
    <Popover.Content container={overlayRef?.current ?? undefined}>
        Rendered in the external overlay root
    </Popover.Content>
</Popover.Root>
```

### Controlled Popover

Most of the time, you'll want to control the popover's open state with `useState` and `open` / `onOpenChange`.

<Wrapper>
	<ControlledPopover />
</Wrapper>

```tsx
export function ControlledPopover() {
    const [open, setOpen] = useState(false);

    const handleDelete = useCallback(() => {
        /**
         * Handle delete logic...
         */
        setOpen(false);
    }, []);

    return (
        <Popover.Root
            hasArrow
            open={open}
            onOpenChange={(nextOpen) => setOpen(nextOpen)}
        >
            <Popover.Trigger>
                <Button variant={"primary"}>Open Popover</Button>
            </Popover.Trigger>

            <Popover.Content>
                <Popover.CloseButton />
                <Popover.Header>Delete Post</Popover.Header>
                <Popover.Body>
                    Are you sure you want to delete this post? This action cannot be undone.
                </Popover.Body>
                <Popover.Footer>
                    <Button variant={"solid"} onClick={() => setOpen(false)}>
                        Cancel
                    </Button>
                    <Button variant={"primary"} onClick={handleDelete}>
                        Delete
                    </Button>
                </Popover.Footer>
            </Popover.Content>
        </Popover.Root>
    );
}
```

### Initial Focus

You can set the initial focus element using `initialFocusRef` prop.

<Wrapper>
	<FocusPopover />
</Wrapper>

```tsx
export function FocusPopover() {
    const [open, setOpen] = useState(false);
    const initialFocusRef = useRef<HTMLButtonElement>(null);

    return (
        <Popover.Root
            hasArrow
            open={open}
            onOpenChange={(nextOpen) => setOpen(nextOpen)}
            initialFocusRef={initialFocusRef}
        >
            <Popover.Trigger>
                <Button variant={"primary"}>Open Popover</Button>
            </Popover.Trigger>

            <Popover.Content>
                <Popover.CloseButton />
                <Popover.Header>Delete Post</Popover.Header>
                <Popover.Body>
                    Are you sure you want to delete this post? This action cannot be undone.
                </Popover.Body>
                <Popover.Footer>
                    <Button variant={"solid"} onClick={() => setOpen(false)} ref={initialFocusRef}>
                        Cancel
                    </Button>
                    <Button variant={"primary"}>Delete</Button>
                </Popover.Footer>
            </Popover.Content>
        </Popover.Root>
    );
}
```

### Placement

You can customize the placement of the popover relative to the trigger element using the `positioning` prop.

<Wrapper>
	<PlacementPopovers />
</Wrapper>

```tsx
const placements = [
    "top", "bottom", "left", "right",
    "top-start", "top-end", "bottom-start", "bottom-end",
    "left-start", "left-end", "right-start", "right-end"
];

export function PlacementPopovers() {
    return (
        <Flex wrapped gap={5}>
            {placements.map((placement) => (
                <PlacementPopover key={placement} placement={placement} />
            ))}
        </Flex>
    );
}

export function PlacementPopover({ placement }: { placement: string }) {
    return (
        <Popover.Root hasArrow positioning={{ placement }}>
            <Popover.Trigger>
                <Button variant={"primary"}>{placement}</Button>
            </Popover.Trigger>

            <Popover.Content>
                <Popover.CloseButton />
                <Popover.Header>Delete Post</Popover.Header>
                <Popover.Body>
                    Are you sure you want to delete this post? This action cannot be undone.
                </Popover.Body>
                <Popover.Footer>
                    <Button variant={"primary"}>Delete</Button>
                </Popover.Footer>
            </Popover.Content>
        </Popover.Root>
    );
}
```

### Size

Use `size` prop to set the size of the popover.

<Wrapper>
	<SizePopovers />
</Wrapper>

```tsx
export function SizePopovers() {
    return (
        <Flex
            wrapped
            gap={5}
        >
            {(
                ["sm", "md", "lg", "xl", "2xl", "3xl", "4xl", "5xl", "6xl", "7xl", "8xl"]
            ).map((size) => (
                <SizePopover key={size} size={size} />
            ))}
        </Flex>
    );
}

export function SizePopover({ size }: { size: string }) {
    return (
        <Popover.Root hasArrow size={size as any}>
            <Popover.Trigger>
                <Button
                    variant={"primary"}
                    w="fit-content"
                >
                    {size}
                </Button>
            </Popover.Trigger>

            <Popover.Content>
                <Popover.CloseButton />
                <Popover.Header>Delete Post</Popover.Header>
                <Popover.Body>
                    Are you sure you want to delete this post? This action cannot be undone.
                </Popover.Body>
                <Popover.Footer>
                    <Button variant={"primary"}>Delete</Button>
                </Popover.Footer>
            </Popover.Content>
        </Popover.Root>
    );
}
```
